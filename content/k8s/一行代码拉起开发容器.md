```bash
# mysql
# empty password
docker run --name mysql -p 3306:3306 -e MYSQL_ALLOW_EMPTY_PASSWORD=yes -d mysql:8
# 个人修改版本. 仅保留 innodb 引擎和服务运行必要的文件
docker run --name mysql -p 3306:3306 -e MYSQL_ALLOW_EMPTY_PASSWORD=yes -d lurenyang/mysql-innodb-server:8.4.11
# password
docker run --name mysql -p 3306:3306 -e MYSQL_ROOT_PASSWORD=<your-password> -d mysql:8
# 同
docker run --name mysql -p 3306:3306 -e MYSQL_ROOT_PASSWORD=<your-password> -d lurenyang/mysql-innodb-server:8.4.11

# redis
docker run -d --name redis -p 6379:6379 redis:8-alpine redis-server --requirepass <your-password>

# nats
docker run --restart=always -d  --name nats \
 -p 4222:4222 \
 -p 8222:8222 \
 nats --http_port 8222

# clickhouse
docker run -d --name ch --restart=always -p 8123:8123 -p 9000:9000 -e CLICKHOUSE_ADMIN_PASSWORD=ch123456 bitnami/clickhouse:25

# etcd
docker run --restart=always -d --name etcd \
    --publish 2379:2379 \
    --publish 2380:2380 \
    --env ALLOW_NONE_AUTHENTICATION=yes \
    --env ETCD_ADVERTISE_CLIENT_URLS=http://127.0.0.1:2379 \
    bitnami/etcd:latest

# mongo
# -e ALLOW_EMPTY_PASSWORD=yes 密码为空
docker run --restart=always -d --name mongo \
  --publish 27017:27017 \
  -e MONGO_INITDB_ROOT_USERNAME=root \
  -e MONGO_INITDB_ROOT_PASSWORD=root123456 \
  mongo

# S3
# silo
docker run \
   --restart always \
   -p 9000:9000 \
   -p 9001:9001 \
   --name silo \
   -e "MINIO_UPDATE=off" \
   -e "MINIO_ROOT_USER=ROOTNAME" \
   -e "MINIO_ROOT_PASSWORD=CHANGEME123" \
   pgsty/silo:distroless server /data --console-address ":9001"

# rustfs
docker run -d \
  --name s3 \
  --restart unless-stopped \
  -p 9000:9000 \
  -p 9001:9001 \
  -e RUSTFS_ACCESS_KEY="ROOTNAME" \
  -e RUSTFS_SECRET_KEY="CHANGEME123" \
  -e RUSTFS_ADDRESS=":9000" \
  -e RUSTFS_CONSOLE_ADDRESS=":9001" \
  -e RUSTFS_CONSOLE_ENABLE=true \
  -e RUSTFS_OBS_LOGGER_LEVEL=error \
  rustfs/rustfs:latest \
  /data
```
